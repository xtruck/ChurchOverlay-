/**
 * detector.js
 * Analyse une phrase et retourne la première référence biblique détectée.
 * Format retourné : { book, chapter, verseStart, verseEnd } ou null.
 * Exemples reconnus : "Jean 3:16", "1 Corinthiens 13:4-7", "Apocalypse 21:4"
 */
function detect(text) {
  if (!text) return null;
  const clean = text.replace(/\[.*?\]/g, '').trim();
  if (!clean) return null;

  const regex = /(\d?\s?[A-Za-zÀ-ÿ]+)\s+(\d+):(\d+)(?:-(\d+))?/g;
  const match = regex.exec(clean);
  if (!match) return null;

  return {
    book: match[1].trim(),
    chapter: parseInt(match[2], 10),
    verseStart: parseInt(match[3], 10),
    verseEnd: match[4] ? parseInt(match[4], 10) : null,
  };
}

module.exports = { detect };
